func run(args []string, in io.Reader, out io.Writer) error {
        fs := flag.NewFlagSet("buddy", flag.ContinueOnError)
        user := fs.String("user", "intern.sokolov", "логин тестовой учётки")
        if err := fs.Parse(args); err != nil {
                return err
        }
        a, err := app.New()
        if err != nil {
                return err
        }
        fmt.Fprintf(out, "Модель: %s/%s (%s), пользователь: %s\n", a.LLM.Provider, a.LLM.Model, a.LLM.Mode, *user)
        reader := bufio.NewReader(in)
        ask := func(text string) error {
                reply, err := a.Agent.Ask(context.Background(), *user, text)
                if err != nil {
                        return err
                }
                show(out, reader, a.Agent, *user, reply)
                return nil
        }
        if fs.NArg() > 0 {
                return ask(strings.Join(fs.Args(), " "))
        }
        for {
                fmt.Fprint(out, "\nТы: ")
                line, _ := reader.ReadString('\n')
                if line = strings.TrimSpace(line); line == "" {
                        return nil
                }
                if err := ask(line); err != nil {
                        return err
                }
        }
}

func show(out io.Writer, in *bufio.Reader, a *agent.Agent, user string, r agent.Reply) {
        fmt.Fprintf(out, "\n🐾 %s\n", r.Text)
        for _, s := range r.Sources {
                fmt.Fprintf(out, "   📎 %s — %s, %s %s\n", s.Title, s.Author, s.UpdatedAt, s.Label)
        }
        for _, d := range r.Drafts {
                fmt.Fprintf(out, "   📝 Черновик %s: %s %v\n   Подтвердить? [y/N] ", d.ID, d.Tool, d.Args)
                line, _ := in.ReadString('\n')
                if strings.TrimSpace(strings.ToLower(line)) == "y" {
                        res := a.Confirm(user, d.ID)
                        if res.Error != nil {
                                fmt.Fprintf(out, "   ❌ %s\n", res.Error.Message)
                        } else {
                                fmt.Fprintf(out, "   ✅ %v\n", res.Receipt)
                        }
                }
        }
}

func (a *Agent) contextFor(u *mockdata.User) string {

// Ask обрабатывает сообщение пользователя.
func (a *Agent) Ask(ctx context.Context, login, text string) (Reply, error) {
        u, found := a.Store.UserByLogin(login)
        if !found {
                return Reply{}, fmt.Errorf("нет пользователя %q", login)
        }
        if answer, blocked := guard(text); blocked {
                return Reply{Text: answer, Drafts: []Draft{}, Sources: []Source{}}, nil
        }
        tc := tools.Ctx{Store: a.Store, User: u}
        var specs []llm.ToolSpec
        for _, t := range tools.For(u) {
                specs = append(specs, llm.ToolSpec{Name: t.Name, Description: t.Description, Schema: t.Schema})
        }
        ctx, cancel := context.WithTimeout(ctx, a.Timeout)
        defer cancel()

        reply := Reply{Drafts: []Draft{}, Sources: []Source{}}
        a.mu.Lock()
        c := a.sessions[login]
        if c == nil || c.turns >= maxTurns {
                c = &chat{session: a.Provider.Start(SystemPrompt, a.contextFor(u), specs)}
                a.sessions[login] = c
        }
        c.turns++
        a.mu.Unlock()
        session := c.session
        step, err := session.SendUser(ctx, text)
        for round := 0; err == nil && len(step.ToolCalls) > 0; round++ {
                if round == maxToolRounds {
                        step = llm.Step{Text: "Не получилось довести запрос до конца. Попробуй сформулировать иначе."}
                        break
                }
                results := make([]llm.ToolResult, 0, len(step.ToolCalls))
                for _, call := range step.ToolCalls {
                        results = append(results, llm.ToolResult{Call: call, Content: a.handle(tc, call, &reply)})
                }
                step, err = session.SendToolResults(ctx, results)
        }
        if err != nil {
                // История могла оборваться посреди вызова инструментов — начинаем разговор заново.
                a.Reset(login)
                return a.fallback(tc, text, err), nil
        }
        reply.Text = step.Text
        if step.Refused {
                reply.Text = "С этим запросом помочь не могу."
        }
        return reply, nil
}

func (a *Agent) handle(tc tools.Ctx, call llm.ToolCall, reply *Reply) string {
        reply.ToolLog = append(reply.ToolLog, call.Name)
        t, known := tools.ByName(call.Name)
        var res any
        if known && t.Confirm && tools.Allowed(t, tc.User) {
                if e := tools.Check(tc, call.Name, call.Input); e != nil {
                        raw, _ := json.Marshal(tools.Result{Error: e})
                        return string(raw)
                }
                d := Draft{ID: a.Store.NextID("DRAFT"), UserID: tc.User.ID, Tool: call.Name, Args: call.Input}
                a.mu.Lock()
                a.drafts[d.ID] = d
                a.mu.Unlock()
                reply.Drafts = append(reply.Drafts, d)
                res = map[string]any{"ok": true, "status": "needs_confirmation", "draft_id": d.ID, "draft": call.Input}
        } else {
                r := tools.Run(tc, call.Name, call.Input)
                res = r
                if call.Name == "search_kb" && r.OK {
                        for _, h := range r.Data.(map[string]any)["hits"].([]tools.Hit) {
                                reply.Sources = append(reply.Sources, Source{h.PageID, h.Title, h.Author, h.UpdatedAt, h.Label})
                        }
                }
        }
        raw, _ := json.Marshal(res)
        return string(raw)
}

// fallback — BR-70 и режим stub_error: модель недоступна → найденные фрагменты базы без пересказа.
func (a *Agent) fallback(tc tools.Ctx, text string, err error) Reply {
        reply := Reply{Drafts: []Draft{}, Sources: []Source{}, Degraded: true, Retry: true}
        reason := "Модель сейчас недоступна."
        if ctxErr := context.DeadlineExceeded; strings.Contains(err.Error(), ctxErr.Error()) {
                reason = "Модель не ответила вовремя."
        }
        hits := tools.SearchKB(tc, text)
        if len(hits) == 0 {
                reply.Text = reason + " Твой вопрос сохранён — нажми «Повторить». План и заявки работают без модели."
                return reply
        }
        reply.Text = reason + " Вот что нашёл в базе без пересказа:\n" + hits[0].Body
        for _, h := range hits {
                reply.Sources = append(reply.Sources, Source{h.PageID, h.Title, h.Author, h.UpdatedAt, h.Label})
        }
        return reply
}

// Reset — начать разговор заново (смена учётки, ошибка модели, кнопка «Новый разговор»).
func (a *Agent) Reset(login string) {
        a.mu.Lock()
        defer a.mu.Unlock()
        delete(a.sessions, login)
}

// Confirm — человек нажал «Подтвердить»: выполнить черновик и вернуть квитанцию (BR-43).
// Если сервис недоступен, черновик сохраняется — можно повторить (SC-15).
func (a *Agent) Confirm(login, draftID string) tools.Result {
        u, found := a.Store.UserByLogin(login)
        a.mu.Lock()
        d, ok := a.drafts[draftID]
        a.mu.Unlock()
        if !found || !ok || d.UserID != u.ID {
                return tools.Result{Error: &tools.Error{Code: "not_found", Message: "Черновик не найден."}}
        }
        res := tools.Run(tools.Ctx{Store: a.Store, User: u}, d.Tool, d.Args)
        if res.Error == nil || res.Error.Code != "unavailable" {
                a.mu.Lock()
                delete(a.drafts, draftID)
                a.mu.Unlock()
        }
        return res
}

// Draft возвращает черновик (для интерфейса).
func (a *Agent) Draft(id string) (Draft, bool) {
        a.mu.Lock()
        defer a.mu.Unlock()
        d, ok := a.drafts[id]
        return d, ok
}

// Проверки ДО модели: такие сообщения вообще не уходят во внешний LLM.

// BR-61, BR-72: чувствительные ПДн и финансы не обрабатываются и не передаются в LLM.
var piiPatterns = []*regexp.Regexp{
        regexp.MustCompile(`\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b`), // номер карты
        regexp.MustCompile(`\b\d{2}\s?\d{2}\s?\d{6}\b`),                  // серия и номер паспорта
        regexp.MustCompile(`\b\d{3}-\d{3}-\d{3}\s?\d{2}\b`),              // СНИЛС
        regexp.MustCompile(`(?i)паспорт|номер карты|реквизит|снилс|диагноз|банковск`),
}

// BR-60: оценка, выход в штат, зарплата, конфликты — решают люди.
var boundaryPattern = regexp.MustCompile(`(?i)в штат|оценк[аиуе]|зарплат|уволят|конфликт|повышени`)

const (
        piiAnswer = "Такие данные я не обрабатываю и никуда не сохраняю. Паспортные и банковские данные вноси только в HR-сервисе — " +
                "там они защищены. Если нужна помощь с заявкой, напиши, что именно оформить, без самих данных."
        boundaryAnswer = "Это решают люди, а не я. Лучше обсудить с бадди или руководителем — могу подготовить вопрос, " +
                "а отправишь его ты."
)

func containsPII(text string) bool {
        for _, re := range piiPatterns {
                if re.MatchString(text) {
                        return true
                }
        }
        return false
}

// MaskPII — для логов и журнала событий (BR-72).
func MaskPII(text string) string {
        for _, re := range piiPatterns[:3] {
                text = re.ReplaceAllString(text, "***")
        }
        return text
}

// guard возвращает готовый ответ, если сообщение нельзя отправлять в модель.
func guard(text string) (string, bool) {
        switch {
        case containsPII(text):
                return piiAnswer, true
        case boundaryPattern.MatchString(strings.ToLower(text)):
                return boundaryAnswer, true
        }
        return "", false
}

// LoadDotEnv читает KEY=VALUE из первого найденного файла и выставляет переменные,
// которые ещё не заданы в окружении (окружение важнее файла). Нужно для запуска без Docker:
// ключ, вписанный в .env в корне репозитория, подхватывается и при `go run`.
func LoadDotEnv(paths ...string) string {
        for _, p := range paths {
                f, err := os.Open(p)
                if err != nil {
                        continue
                }
                sc := bufio.NewScanner(f)
                for sc.Scan() {
                        line := strings.TrimSpace(sc.Text())
                        if line == "" || strings.HasPrefix(line, "#") {
                                continue
                        }
                        key, val, ok := strings.Cut(line, "=")
                        key, val = strings.TrimSpace(key), strings.Trim(strings.TrimSpace(val), `"'`)
                        if _, set := os.LookupEnv(key); ok && key != "" && !set {
                                _ = os.Setenv(key, val)
                        }
                }
                _ = f.Close()
                return p
        }
        return ""
}

                })
        }
        return mux
}

type handler func(http.ResponseWriter, *http.Request, *mockdata.User)

func (s *Server) auth(h handler) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                u, ok := s.Store.UserByLogin(r.Header.Get("X-User"))
                if !ok {
                        writeJSON(w, http.StatusUnauthorized, errBody("unauthorized", "Укажи тестовую учётку в заголовке X-User"))
                        return
                }
                h(w, r, u)
        }
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

func errBody(code, msg string) map[string]any {
        return map[string]any{"ok": false, "error": map[string]string{"code": code, "message": msg}}
}

var statusByCode = map[string]int{"forbidden": 403, "not_found": 404, "validation": 422, "unavailable": 503}

func writeResult(w http.ResponseWriter, res tools.Result) {
        status := http.StatusOK
        if res.Error != nil {
                status = statusByCode[res.Error.Code]
                if status == 0 {
                        status = http.StatusBadRequest
                }
        }
        writeJSON(w, status, res)
}

// Anthropic — Claude через официальный Go SDK. Ключ: LLM_API_KEY или ANTHROPIC_API_KEY.
//
// Стабильная часть (инструкции и инструменты) кэшируется — повторные запросы дешевле.
// Контекст пользователя — отдельным блоком после кэшируемой части.
type Anthropic struct {
        client *anthropic.Client
        cfg    Config
}

func NewAnthropic(c Config, opts ...option.RequestOption) *Anthropic {
        if c.APIKey != "" {
                opts = append(opts, option.WithAPIKey(c.APIKey))
        }
        client := anthropic.NewClient(opts...)
        return &Anthropic{client: &client, cfg: c}
}

func (a *Anthropic) Start(system, context string, tools []ToolSpec) Session {
        var ts []anthropic.ToolUnionParam
        for _, t := range tools {
                tp := anthropic.ToolParam{
                        Name:        t.Name,
                        Description: anthropic.String(t.Description),
                        InputSchema: anthropic.ToolInputSchemaParam{Properties: t.Schema["properties"], Required: toStrings(t.Schema["required"])},
                }
                ts = append(ts, anthropic.ToolUnionParam{OfTool: &tp})
        }
        return &anthropicSession{a: a, tools: ts, system: []anthropic.TextBlockParam{
                {Text: system, CacheControl: anthropic.NewCacheControlEphemeralParam()},
                {Text: context},
        }}
}

type anthropicSession struct {
        a        *Anthropic
        tools    []anthropic.ToolUnionParam
        system   []anthropic.TextBlockParam
        messages []anthropic.MessageParam
}

func (s *anthropicSession) call(ctx context.Context) (Step, error) {
        resp, err := s.a.client.Messages.New(ctx, anthropic.MessageNewParams{
                Model:        s.a.cfg.Model,
                MaxTokens:    16000,
                System:       s.system,
                Tools:        s.tools,
                Messages:     s.messages,
                OutputConfig: anthropic.OutputConfigParam{Effort: anthropic.OutputConfigEffort(s.a.cfg.Effort)},
        })
        if err != nil {
                return Step{}, err
        }
        // Ответ модели (включая блоки размышлений) возвращаем в историю без изменений
        s.messages = append(s.messages, resp.ToParam())
        step := Step{Refused: resp.StopReason == anthropic.StopReasonRefusal}
        for _, block := range resp.Content {
                switch b := block.AsAny().(type) {
                case anthropic.TextBlock:
                        step.Text += b.Text
                case anthropic.ToolUseBlock:
                        input := map[string]any{}
                        _ = json.Unmarshal([]byte(b.JSON.Input.Raw()), &input)
                        step.ToolCalls = append(step.ToolCalls, ToolCall{ID: b.ID, Name: b.Name, Input: input})
                }
        }
        return step, nil
}

func (s *anthropicSession) SendUser(ctx context.Context, text string) (Step, error) {
        s.messages = append(s.messages, anthropic.NewUserMessage(anthropic.NewTextBlock(text)))
        return s.call(ctx)
}

func (s *anthropicSession) SendToolResults(ctx context.Context, results []ToolResult) (Step, error) {
        // Все результаты — одним сообщением, иначе модель перестаёт вызывать инструменты параллельно
        blocks := make([]anthropic.ContentBlockParamUnion, 0, len(results))
        for _, r := range results {
                blocks = append(blocks, anthropic.NewToolResultBlock(r.Call.ID, r.Content, false))
        }
        s.messages = append(s.messages, anthropic.NewUserMessage(blocks...))
        return s.call(ctx)
}

func toStrings(v any) []string {
        switch xs := v.(type) {
        case []string:
                return xs
        case []any:
                out := make([]string, 0, len(xs))
                for _, x := range xs {
                        if s, ok := x.(string); ok {
                                out = append(out, s)
                        }
                }
                return out
        }
        return nil
}

func ConfigFromEnv() (Config, error) {
        c := Config{
                Provider: strings.ToLower(strings.TrimSpace(os.Getenv("LLM_PROVIDER"))),
                Model:    os.Getenv("LLM_MODEL"),
                APIKey:   os.Getenv("LLM_API_KEY"),
                BaseURL:  os.Getenv("LLM_BASE_URL"),
                Mode:     os.Getenv("LLM_MODE"),
                Effort:   os.Getenv("LLM_EFFORT"),
                Timeout:  30 * time.Second,
        }
        if c.Provider == "claude" {
                c.Provider = "anthropic"
        }
        // «Вставил ключ — заработало»: провайдер не указан, но есть ключ → включаем соответствующую модель.
        if c.Provider == "" && !strings.HasPrefix(c.Mode, "stub") {
                switch {
                case c.APIKey != "" || os.Getenv("ANTHROPIC_API_KEY") != "":
                        c.Provider = "anthropic"
                case os.Getenv("DEEPSEEK_API_KEY") != "":
                        c.Provider = "deepseek"
                }
        }
        if c.Mode == "" {
                c.Mode = "live"
        }
        if c.Provider == "" || strings.HasPrefix(c.Mode, "stub") {
                c.Provider = "stub"
                if c.Mode == "live" {
                        c.Mode = "stub_ok"
                }
        }
        if _, known := defaultModels[c.Provider]; !known {
                return c, fmt.Errorf("LLM_PROVIDER=%q: допустимо anthropic, deepseek или stub", c.Provider)
        }
        if c.Model == "" {
                c.Model = defaultModels[c.Provider]
        }
        if c.Effort == "" {
                c.Effort = "medium"
        }
        if v, err := strconv.Atoi(os.Getenv("LLM_TIMEOUT_SEC")); err == nil && v > 0 {
                c.Timeout = time.Duration(v) * time.Second
        }
        if c.APIKey == "" {
                c.APIKey = map[string]string{"anthropic": os.Getenv("ANTHROPIC_API_KEY"), "deepseek": os.Getenv("DEEPSEEK_API_KEY")}[c.Provider]
        }
        return c, nil
}

func (s *stubSession) wait(ctx context.Context) error {

        case "stub_error":

                        sec = 15
                }
                select {
                case <-time.After(time.Duration(sec) * time.Second):

        }
        return nil
}

var isoDate = regexp.MustCompile(`\d{4}-\d{2}-\d{2}`)

var weekdays = map[string]time.Weekday{"понедельник": time.Monday, "вторник": time.Tuesday, "сред": time.Wednesday,
        "четверг": time.Thursday, "пятниц": time.Friday}

// ParseDay — дата из фразы: ISO, «завтра», «в пятницу» (ближайшая следующая).
func ParseDay(text string, today time.Time) string {
        if m := isoDate.FindString(text); m != "" {
                return m
        }
        low := strings.ToLower(text)
        if strings.Contains(low, "завтра") {
                return today.AddDate(0, 0, 1).Format(time.DateOnly)
        }
        for stem, wd := range weekdays {
                if strings.Contains(low, stem) {
                        delta := (int(wd) - int(today.Weekday()) + 7) % 7
                        if delta == 0 {
                                delta = 7
                        }
                        return today.AddDate(0, 0, delta).Format(time.DateOnly)
                }
        }
        return ""
}

        for _, r := range results {
                var res stubResult
                _ = json.Unmarshal([]byte(r.Content), &res)
                if !res.OK && res.Error != nil {
                        msg := res.Error.Message
                        if res.Error.WhoCan != "" {
                                msg += " Могу подготовить запрос: " + res.Error.WhoCan + "."
                        }
                        parts = append(parts, msg)
                        continue

                if res.Status == "needs_confirmation" {

                        continue

                text, next := s.describe(r.Call.Name, res.Data)
                if next != nil {
                        return s.filter([]ToolCall{*next}), nil
                }
                if text != "" {
                        parts = append(parts, text)
                }
        }
        if len(parts) == 0 {
                parts = append(parts, "Готово.")
        }
        return Step{Text: strings.Join(parts, "\n")}, nil
}

// MinCoverage — доля слов вопроса, найденных на странице. Ниже порога — «не нашёл», а не слабый ответ (BR-04).
const MinCoverage = 0.6

// ExpertThreshold — BR-13: уверенность выбора эксперта ниже порога → вопрос бадди.
const ExpertThreshold = 0.6

// MinExpertEvidence — сколько «очков» сигнала нужно для полной уверенности (страница по теме ≈ 1–4, зона ответственности ≈ 1.5).
const MinExpertEvidence = 3.0

var wordRe = regexp.MustCompile(`[\p{L}\p{N}_]+`)

var stopWords = map[string]bool{
        "как": true, "что": true, "где": true, "это": true, "мне": true, "для": true, "через": true, "такое": true,
        "нужно": true, "нужен": true, "можно": true, "почему": true, "когда": true, "какой": true, "какие": true,
        "кто": true, "мой": true, "или": true, "если": true, "есть": true, "наш": true, "нашем": true, "the": true,
}

// synonyms — словарь синонимов отдела: сводит разные слова к одной основе.
// Данные, а не логика: в проде — из настроек отдела. Пополняется по эталонному набору evals.json.
var synonyms = map[string]string{
        "отлич": "расхо", // «отличается» ≈ «расхождение»
        "совпа": "расхо", // «не совпадает» ≈ «расхождение»
        "сходи": "расхо",
}

// Stems — грубый стемминг префиксом: для русского работает прилично и не требует словарей.
func Stems(text string) map[string]bool {
        out := map[string]bool{}
        for _, w := range wordRe.FindAllString(strings.ToLower(strings.ReplaceAll(text, "ё", "е")), -1) {
                r := []rune(w)
                if len(r) < 3 || stopWords[w] {
                        continue
                }
                if len(r) > 5 {
                        r = r[:5]
                }
                stem := string(r)
                if syn, found := synonyms[stem]; found {
                        stem = syn
                }
                out[stem] = true
        }
        return out
}

func overlap(q, doc map[string]bool) int {
        n := 0
        for s := range q {
                if doc[s] {
                        n++
                }

        return n
}

// FindExpert — BR-10…BR-13: кандидат по авторству страниц по теме (свежие весят больше) и зоне ответственности.
//
// Базовый лексический подбор. Лидер на сценарных вопросах верный (Q2 → Орлов, Q3 → Смирнова, Q5 → Лебедева),
// но уверенность у Q2 и Q5 ниже порога BR-13 → они уйдут бадди. Настраивать по docs/04-mock-data/evals.json.
func FindExpert(c Ctx, question string) (id string, confidence float64, reason string) {
        q := Stems(question)
        type cand struct {
                pageScores []float64
                resp       int
        }
        cands := map[string]*cand{}
        get := func(uid string) *cand {
                if cands[uid] == nil {
                        cands[uid] = &cand{}
                }
                return cands[uid]
        }
        for _, p := range readablePages(c) {
                n := overlap(q, Stems(p.Title+" "+p.Body))
                if p.AuthorID == "" || p.AuthorID == c.User.ID || n == 0 {
                        continue
                }
                get(p.AuthorID).pageScores = append(get(p.AuthorID).pageScores, recencyWeight(c, p.UpdatedAt)*float64(n))
        }
        for _, u := range c.Store.Data.Team.Users {
                if u.ID == c.User.ID || store.HasRole(&u, store.RoleIntern) {
                        continue
                }
                if n := overlap(q, Stems(u.Responsibility)); n > 0 {
                        get(u.ID).resp = n
                }
        }
        score := func(cd *cand) float64 {
                ps := append([]float64(nil), cd.pageScores...)
                sort.Sort(sort.Reverse(sort.Float64Slice(ps)))
                total := 1.5 * float64(cd.resp)
                for i := 0; i < len(ps) && i < 2; i++ { // две лучшие страницы: число страниц само по себе не экспертиза
                        total += ps[i]
                }
                return total
        }
        var best, second float64
        for uid, cd := range cands {
                if sc := score(cd); sc > best {
                        second, best, id = best, sc, uid
                } else if sc > second {
                        second = sc
                }
        }
        if id == "" {
                return "", 0, ""
        }
        // Уверенность: насколько лидер отрывается от второго, с поправкой на силу сигнала
        // (одно случайное совпадение слова — ещё не экспертиза).
        confidence = best / (best + second) * min(1, best/MinExpertEvidence)
        cd := cands[id]
        pages := len(cd.pageScores)
        switch {
        case pages > 0 && cd.resp > 0:
                reason = fmt.Sprintf("автор страниц по теме (%d) и это его зона ответственности", pages)
        case pages > 0:
                reason = fmt.Sprintf("автор страниц по теме (%d)", pages)
        default:
                reason = "это его зона ответственности"
        }
        return id, confidence, reason
}

func findExpert(c Ctx, args map[string]any) Result {
        id, conf, reason := FindExpert(c, str(args, "question"))
        if id == "" || conf < ExpertThreshold {
                buddy := c.User.BuddyID
                return ok(map[string]any{"addressee_id": buddy, "name": c.Store.Name(buddy), "is_buddy": true,
                        "reason": "уверенного эксперта нет — спросим бадди", "confidence": conf})
        }
        return ok(map[string]any{"addressee_id": id, "name": c.Store.Name(id), "is_buddy": false, "reason": reason, "confidence": conf})
}

// recencyWeight — BR-10: свежие страницы весят больше, давно не обновлённые — слабый сигнал.
func recencyWeight(c Ctx, updatedAt string) float64 {
        upd, err := time.Parse(time.DateOnly, updatedAt)
        if err != nil {
                return 1
        }
        age := c.Store.Today.Sub(upd)
        switch {
        case age <= 30*24*time.Hour:
                return 2
        case age <= 180*24*time.Hour:
                return 1
        case age <= 365*24*time.Hour:
                return 0.6
        default:
                return 0.3
        }
}

func targetIntern(c Ctx, internID string) (string, *Result) {
        if internID == "" {

func planViews(c Ctx, internID string) []stepView {
        var out []stepView
        for _, st := range c.Store.PlanSteps(internID) {
                t, _ := c.Store.TemplateStep(st.StepID)
                out = append(out, stepView{st.StepID, t.Title, t.Stage, t.Owner, st.DueDate, st.Status, t.Source})
        }
        return out
}

func getPlan(c Ctx, args map[string]any) Result {
        internID, errRes := targetIntern(c, str(args, "intern_id"))
        if errRes != nil {
                return *errRes
        }
        steps := planViews(c, internID)
        if len(steps) == 0 {
                return fail("not_found", "План адаптации ещё не утверждён.")
        }
        var mine, done int
        var next *stepView
        for i, s := range steps {
                if s.Owner != store.RoleIntern {
                        continue
                }
                mine++
                if s.Status == "Done" {
                        done++
                } else if next == nil || s.Status == "Overdue" && next.Status != "Overdue" {
                        next = &steps[i]
                }
        }
        return ok(map[string]any{
                "intern": c.Store.Name(internID), "stages": c.Store.Data.Template.Stages, "steps": steps,
                "progress_pct": done * 100 / max(mine, 1), "next_step": next,
        })
}

func getToday(c Ctx, _ map[string]any) Result {
        var overdue, today []stepView
        for _, s := range planViews(c, c.User.ID) {
                if s.Owner != store.RoleIntern {
                        continue
                }
                switch s.Status {
                case "Overdue":
                        overdue = append(overdue, s)
                case "Today":
                        today = append(today, s)
                }
        }
        steps := append(overdue, today...)
        if len(steps) > 3 {
                steps = steps[:3]
        }
        day := c.Store.Today.Format("2006-01-02")
        var events []map[string]string
        for _, e := range c.Store.Data.Calendar.Events {
                if strings.HasPrefix(e.Start, day) && containsStr(e.Participants, c.User.ID) {
                        events = append(events, map[string]string{"title": e.Title, "start": e.Start[11:16]})
                }
        }
        return ok(map[string]any{"week": c.Store.Week(c.User), "steps": steps, "events": events})
}

func markStepDone(c Ctx, args map[string]any) Result {
        internID := str(args, "intern_id")
        if internID == "" {
                internID = c.User.ID
        }
        if internID != c.User.ID {

                if !found || intern.BuddyID != c.User.ID {
                        return fail("forbidden", "Отметить шаг может стажёр или его бадди.")

        stepID := str(args, "step_id")
        if !c.Store.SetStepStatus(internID, stepID, "Done") {
                return fail("not_found", "Такого шага нет в плане.")
        }
        return Result{OK: true, Data: map[string]string{"step_id": stepID, "status": "Done"},
                Receipt: map[string]string{"id": "STEP-" + stepID, "status": "Done"}}
}

}

                ws = append(ws, warning{"SHORT_NOTICE", "По тестовой инструкции отгул подают за 2 рабочих дня"})

        date := day.Format(time.DateOnly)
        for _, e := range c.Store.Data.Calendar.Events {
                if strings.HasPrefix(e.Start, date) && containsStr(e.Participants, c.User.ID) && e.Type != "deadline" {
                        ws = append(ws, warning{"MEETING_CONFLICT", "В этот день у тебя встреча: " + e.Title})
                }
        }
        for _, s := range planViews(c, c.User.ID) {
                if s.DueDate == date && s.Status != "Done" && s.Owner == store.RoleIntern {
                        ws = append(ws, warning{"STEP_DEADLINE", "В этот день срок шага плана: " + s.Title})
                }
        }
        return ok(map[string]any{"date": date, "warnings": ws})

// Run выполняет инструмент с проверкой прав (BR-40).

        if !found || !Allowed(t, c.User) {
                return fail("forbidden", "Это действие тебе недоступно.")
        }
        if args == nil {
                args = map[string]any{}
        }
        return t.Run(c, args)
}

func parseDate(s string) (time.Time, bool) {
        t, err := time.Parse(time.DateOnly, s)
        return t, err == nil
}

func workdaysBetween(from, to time.Time) int {
        n := 0
        for d := from.AddDate(0, 0, 1); !d.After(to); d = d.AddDate(0, 0, 1) {
                if d.Weekday() != time.Saturday && d.Weekday() != time.Sunday {
                        n++
                }
        }
        return n
}

// Check — проверки прав и данных для инструментов с подтверждением, БЕЗ побочных эффектов.
// Агент вызывает их до создания черновика: недоступное не предлагается даже черновиком (BR-40, BR-44).

        if !found || !Allowed(t, c.User) {

        }
        switch name {
        case "request_access":

        case "send_question":
                if _, ok := c.Store.User(str(args, "addressee_id")); !ok {
                        return &Error{Code: "not_found", Message: "Адресат не найден в команде."}
                }
        case "create_absence_request":
                if _, ok := requestType(c, str(args, "type")); !ok {
                        return &Error{Code: "validation", Message: "Неизвестный тип отсутствия."}
                }
                if day, ok := parseDate(str(args, "date")); !ok || !day.After(c.Store.Today) {
                        return &Error{Code: "validation", Message: "Укажи будущую дату"}
                }
        case "mark_step_done":
                if internID := str(args, "intern_id"); internID != "" && internID != c.User.ID {
                        if intern, ok := c.Store.User(internID); !ok || intern.BuddyID != c.User.ID {
                                return &Error{Code: "forbidden", Message: "Отметить шаг может стажёр или его бадди."}
                        }
                }
        }
        return nil
}
